## Implementasi
```vue
<template>
    <div>
        <h5 class="mb-4">Silahkan masukkan NIK Anda untuk memulai aplikasi.</h5>
        <div class="form-group d-flex align-items-center">
            <input type="text" class="form-control mr-2" id="nik" placeholder="Input NIK" v-model="nik">&nbsp;&nbsp;&nbsp;
            <button class="btn btn-success" @click="login">LOGIN</button>
        </div>
    </div>
</template>

<script setup>
import { ref } from 'vue';
import Swal from 'sweetalert2';
import { useRouter } from 'vue-router';
import { getDatabase, ref as dbRef, get, child } from 'firebase/database';

const nik = ref('');
const router = useRouter();

async function login() {
  if (!nik.value) {
    Swal.fire('Error', 'NIK tidak boleh kosong!', 'error');
    return;
  }

  const db = getDatabase();
  const dbReference = dbRef(db);

  try {
    const snapshot = await get(child(dbReference, 'sms_driver'));

    if (snapshot.exists()) {
      const users = snapshot.val();
      const userExists = Object.values(users).find((user) => user.nik === nik.value);

      if (userExists) {
        const nama = userExists.nama;
        router.push({ path: '/home', query: { nik: nik.value, nama } });
      } else {
        Swal.fire('Gagal', 'NIK tidak terdaftar di sistem!', 'error');
      }
    } else {
      Swal.fire('Gagal', 'NIK tidak terdaftar di sistem!', 'error');
    }
  } catch (error) {
    console.error(error);
    Swal.fire('Error', 'Terjadi kesalahan saat memeriksa NIK!', 'error');
  }
}
</script>

<style scoped>
.form-group {
  display: flex;
  align-items: center;
}
</style>
```

Penjelasan:
- **`ref`:** Digunakan untuk membuat variabel reaktif (`nik`) yang dapat diikat ke input menggunakan `v-model`.
- **`Swal`:** SweetAlert2 digunakan untuk menampilkan notifikasi atau pesan popup kepada pengguna.
- **`useRouter`:** Vue Router digunakan untuk mengarahkan pengguna ke halaman lain setelah login berhasil.
- `getDatabase`: Mendapatkan instance database Firebase.
- `ref`: Membuat referensi ke root database.
- `child`: Mengakses node tertentu di database (dalam hal ini, `sms_driver`).
- `get`: Mengambil data dari database secara asynchronous.
- **`nik`:** Variabel reaktif yang menyimpan nilai input NIK dari pengguna.
- **`router`:** Instance Vue Router untuk navigasi.
