## Membuat Controller Manual
Note: lebih banyak main di folder src.
1. Buat file `/biodata/biodata.controller.ts` 
```typescript
import { Controller, Get } from '@nestjs/common';

@Controller('biodata')
export class BiodataController {
  @Get()
  findAll(): string {
    return 'Nama gue Andaru';
  }
}
```

2. Buat file `/biodata/biodata.module.ts`
```typescript
import { Module } from '@nestjs/common';
import { BiodataController } from './biodata.controller';

@Module({
  controllers: [BiodataController],
})

export class BiodataModule {}
```

3. Update kode file `/src/app.module.ts`
```typescript
import { Module } from '@nestjs/common';
import { AppController } from './app.controller';
import { AppService } from './app.service';
import { CatsModule } from './cats.module';
import { BiodataModule } from './biodata/biodata.module';

@Module({
  imports: [CatsModule, BiodataModule],
  controllers: [AppController],
  providers: [AppService],
})

export class AppModule {}
```

Kita akan load module dan controller di appModuleTs

Lalu ketik perintah `npm run start:dev`.

```typescript
import { Controller, Get, Redirect, Param } from '@nestjs/common';

@Controller('biodata')
export class BiodataController {
  @Get() // localhost:3000/biodata // diredirect langsung parent url nya
  @Redirect('https://andarutriadi.netlify.app')

  @Get('nama') // localhost:3000/biodata/nama
  nama(): string {
    return 'Nama gue andaru triadi';
  }

  @Get('/check/:id') // localhost:3000/biodata/check/bebas
  check(@Param('id') id: string): string {
    return `Halo ${id}`;
  }
}
```
